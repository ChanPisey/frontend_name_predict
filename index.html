ទាញយក API

អ្នកអភិវឌ្ឍន៍អាចភ្ជាប់ API នេះទៅកាន់កម្មវិធីរបស់អ្នកបានដោយផ្ញើ សំណើ HTTP មកកាន់ endpoint ខាងក្រោម។

ទស្សន៍ទាយឈ្មោះតែមួយ

POST /predict
Content-Type: application/json

{
  "name": "ចន្ទា"
}

លទ្ធផលដែលទទួលបាន៖

{
  "name": "ចន្ទា",
  "gender": "Female",
  "confidence": 99.91,
  "probability": 0.9991,
  "kccs": ["ច", "ន្ទា"]
}

ទស្សន៍ទាយជាក្រុម (អតិបរមា ១០០ ឈ្មោះ)

POST /batch_predict
Content-Type: application/json

{
  "names": ["ចន្ទា", "សុខ", "រដ្ឋា"]
}

ពិនិត្យស្ថានភាព API

GET /health

ឧទាហរណ៍ជាមួយ JavaScript

const res = await fetch('https://yourdomain.com/predict', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ name: 'ចន្ទា' })
});
const data = await res.json();
console.log(data.gender, data.confidence);

ឧទាហរណ៍ជាមួយ Python

import requests

res = requests.post('https://yourdomain.com/predict',
                    json={'name': 'ចន្ទា'})
print(res.json())

សម្រាប់ឯកសារពេញលេញ សូមចូលទៅកាន់ /docs (Swagger UI) បន្ទាប់ពីដំណើរការ API ។

អំពីគម្រោង

គម្រោងនេះត្រូវបានបង្កើតឡើងដើម្បីជួយកំណត់ភេទតាមរយៈឈ្មោះខ្មែរ ដោយប្រើប្រាស់បញ្ញាសិប្បនិម្មិត។ ឈ្មោះខ្មែរជាច្រើនមានលក្ខណៈពិសេស ដែលអាចបង្ហាញពីភេទបានដោយប្រយោល ប៉ុន្តែការវិភាគដោយដៃច្រើនតែចំណាយ ពេលវេលា និងកំហុសអាចកើតមានបាន។

ដោយប្រើប្រាស់បច្ចេកវិទ្យា AI ដ៏ទំនើប ប្រព័ន្ធអាចទស្សន៍ទាយភេទ ប្រកបដោយភាពត្រឹមត្រូវ និងភាពឆាប់រហ័ស ដោយមិនចាំបាច់ពឹងផ្អែកលើ មនុស្សក្នុងការកំណត់ឡើយ។

គម្រោងនេះមានប្រយោជន៍សម្រាប់ប្រព័ន្ធគ្រប់គ្រងទិន្នន័យ ការស្រាវជ្រាវ ក៏ដូចជាកម្មវិធីដែលត្រូវការចាត់ថ្នាក់ភេទ ពីឈ្មោះអ្នកប្រើប្រាស់ជាភាសាខ្មែរ។
